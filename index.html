Los montos se guardan en Supabase en la tabla recaudacion (separada de los datos del torneo). Inscripción por defecto: USD 40. Multas por fecha: USD 0 por defecto — actualizalas cuando se genere la obligación. Los cambios se sincronizan en tiempo real.

SQL para crear la tabla en Supabase
-- Ejecutar en Supabase → SQL Editor

CREATE TABLE IF NOT EXISTS recaudacion (
  id          TEXT PRIMARY KEY,         -- player_id ej: 'gonzalez'
  nombre      TEXT NOT NULL,
  equipo      TEXT NOT NULL,            -- 'g' | 'r'
  inscripcion NUMERIC(8,2) DEFAULT 40,
  f1  NUMERIC(8,2) DEFAULT 0,
  f2  NUMERIC(8,2) DEFAULT 0,
  f3  NUMERIC(8,2) DEFAULT 0,
  f4  NUMERIC(8,2) DEFAULT 0,
  f5  NUMERIC(8,2) DEFAULT 0,
  f6  NUMERIC(8,2) DEFAULT 0,
  f7  NUMERIC(8,2) DEFAULT 0,
  f8  NUMERIC(8,2) DEFAULT 0,
  f9  NUMERIC(8,2) DEFAULT 0,
  f10 NUMERIC(8,2) DEFAULT 0,
  updated_at  TIMESTAMPTZ DEFAULT NOW()
);

-- Row-level security: lectura pública, escritura solo con service_role o anon
ALTER TABLE recaudacion ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow public read recaudacion"
  ON recaudacion FOR SELECT USING (true);

CREATE POLICY "Allow anon insert/update recaudacion"
  ON recaudacion FOR ALL USING (true) WITH CHECK (true);

CREATE TABLE IF NOT EXISTS fixture (
  num INT PRIMARY KEY,
  fecha TEXT,
  modalidad TEXT,
  criterio TEXT,
  organizadores TEXT,
  updated_at TIMESTAMPTZ DEFAULT NOW()
);
ALTER TABLE fixture ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow public read fixture"
  ON fixture FOR SELECT USING (true);
CREATE POLICY "Allow anon write fixture"
  ON fixture FOR ALL USING (true) WITH CHECK (true);
